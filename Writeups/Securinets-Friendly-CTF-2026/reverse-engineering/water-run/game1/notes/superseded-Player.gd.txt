extends CharacterBody3D

# --- ENDLESS RUNNER: PLAYER MECHANICS ---
# This script handles the core gameplay loop for our hero (Aziz H. / Aziz R.)
# They run down a 3-lane track, dodging obstacles and protecting their water.

const LANE_WIDTH = 2.5
const FORWARD_SPEED = 15.0
const JUMP_VELOCITY = 8.0
const GRAVITY = 20.0

var current_lane: int = 1 # 0: Left, 1: Middle, 2: Right
var target_x: float = 0.0

# The core mechanic: 6 bottles of water instead of a health bar
var water_bottles: int = 6

signal obstacle_hit
signal game_over

func _ready():
    # Initialize position in the middle lane
    target_x = (current_lane - 1) * LANE_WIDTH
    position.x = target_x

func _physics_process(delta):
    # 1. Constant forward movement
    velocity.z = -FORWARD_SPEED
    
    # 2. Handle Gravity
    if not is_on_floor():
        velocity.y -= GRAVITY * delta
        
    # 3. Handle Lane Switching Input
    if Input.is_action_just_pressed("ui_left") and current_lane > 0:
        current_lane -= 1
        target_x = (current_lane - 1) * LANE_WIDTH
    elif Input.is_action_just_pressed("ui_right") and current_lane < 2:
        current_lane += 1
        target_x = (current_lane - 1) * LANE_WIDTH
        
    # Smoothly interpolate X position towards the target lane for a snappy but smooth feel
    position.x = lerp(position.x, target_x, 15.0 * delta)
    
    # 4. Handle Jump
    if Input.is_action_just_pressed("ui_up") and is_on_floor():
        velocity.y = JUMP_VELOCITY
        
    move_and_slide()

# Called when an obstacle's collision box detects the player
func take_hit():
    if water_bottles > 0:
        water_bottles -= 1
        emit_signal("obstacle_hit")
        
        # Give brief invulnerability or a stumble animation trigger here
        
        if water_bottles == 0:
            die()

func die():
    # Emit game over so the CrowdManager knows to engulf the player
    emit_signal("game_over")
    set_physics_process(false)
