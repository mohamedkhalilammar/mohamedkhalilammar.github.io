class_name CGCharacters
extends Node

const HIP_Y := 0.95
const THIGH_LEN := 0.44
const SHIN_LEN := 0.42
const TORSO_LEN := 0.50
const UPPER_ARM := 0.28
const FOREARM := 0.26
const HIP_SPREAD := 0.115
const SHOULDER_SPREAD := 0.215

const CARRY_ARM_PITCH := 0.44
const CARRY_ARM_YAW := 0.10
const CARRY_ELBOW := 1.48
const SWING_ELBOW := 0.55
const TORSO_LEAN := -0.22

const RUN_LEN := 0.56
const HALF := 0.28

const PACK_SLOTS := [
    Vector2(-1, 1), Vector2(-1, 0), Vector2(0, 1),
    Vector2(0, 0), Vector2(1, 1), Vector2(1, 0),
]

const SKINS := [
    Color(0.86, 0.69, 0.53), Color(0.77, 0.58, 0.42),
    Color(0.66, 0.48, 0.34), Color(0.81, 0.63, 0.47),
]

static func make_hero(hero_id: String) -> Node3D:
    var cfg := _hero_config(hero_id)
    var root := _build(cfg)
    root.name = "Body"
    root.set_meta("cg_kind", "hero")
    root.set_meta("cg_hero", cfg["id"])
    set_bottles(root, 6)
    play_state(root, "run")
    return root

static func make_crowd_member(index: int) -> Node3D:
    var cfg := _crowd_config(index)
    var root := _build(cfg)
    root.name = "CrowdMember"
    root.set_meta("cg_kind", "crowd")
    root.set_meta("cg_archetype", cfg["id"])
    var ap: AnimationPlayer = root.get_node("Anim")
    ap.speed_scale = cfg["speed"]
    play_state(root, "run")
    ap.seek(cfg["phase"] * RUN_LEN, true)
    return root

static func play_state(node: Node3D, state: String) -> void:
    if node == null:
        return
    var ap := node.get_node_or_null("Anim") as AnimationPlayer
    if ap == null:
        return
    var want := state.strip_edges().to_lower()
    if not ap.has_animation(want):
        want = "run"
    if not ap.has_animation(want):
        return
    if want in ["run", "idle"] and ap.current_animation == want and ap.is_playing():
        return
    ap.play(want)
    if want in ["stumble", "jump"]:
        ap.queue("run")

static func set_bottles(node: Node3D, n: int) -> void:
    if node == null:
        return
    var pack := node.get_node_or_null("Rig/Hips/Torso/Carry/SixPack")
    if pack == null:
        return
    var keep := clampi(n, 0, 6)
    for i in 6:
        var b := pack.get_node_or_null("Bottle%d" % i) as Node3D
        if b != null:
            b.visible = i < keep
    pack.visible = keep > 0

static func set_transparency(node: Node3D, amount: float) -> void:
    if node == null:
        return
    var a := clampf(amount, 0.0, 1.0)
    for mi in node.find_children("*", "MeshInstance3D", true, false):
        mi.transparency = a

static func _hero_config(hero_id: String) -> Dictionary:
    var burgundy := hero_id.strip_edges().to_lower() == "rahmouni"
    return {
        "id": "rahmouni" if burgundy else "haddadi",
        "skin": SKINS[1] if burgundy else SKINS[2],
        "shirt": Color(0.36, 0.08, 0.15) if burgundy else Color(0.11, 0.11, 0.13),
        "pants": Color(0.13, 0.13, 0.15) if burgundy else Color(0.17, 0.20, 0.30),
        "shoes": Color(0.93, 0.93, 0.90) if burgundy else Color(0.09, 0.09, 0.10),
        "hair": Color(0.09, 0.07, 0.06),
        "beard": not burgundy,
        "glasses": burgundy,
        "cap": Color(0, 0, 0, 0),
        "band": Color(0, 0, 0, 0) if burgundy else Color(0.94, 0.44, 0.62),
        "carry": true,
        "stoop": 0.0,
        "scale": 1.0,
        "width": 1.0,
        "speed": 1.0,
        "phase": 0.0,
    }

static func _crowd_config(index: int) -> Dictionary:
    var rng := RandomNumberGenerator.new()
    rng.seed = index * 7919 + 104729
    var kind := index % 4
    var cfg := {
        "id": "civilian",
        "skin": SKINS[rng.randi() % SKINS.size()],
        "shirt": Color(0.7, 0.7, 0.7),
        "pants": Color(0.28, 0.26, 0.24),
        "shoes": Color(0.18, 0.15, 0.13),
        "hair": Color(0.08, 0.06, 0.05),
        "beard": rng.randf() < 0.35,
        "glasses": rng.randf() < 0.25,
        "cap": Color(0, 0, 0, 0),
        "band": Color(0, 0, 0, 0),
        "carry": false,
        "stoop": 0.0,
        "scale": rng.randf_range(0.93, 1.07),
        "width": rng.randf_range(0.92, 1.14),
        "speed": rng.randf_range(0.86, 1.16),
        "phase": rng.randf(),
    }
    if kind == 0:
        cfg["id"] = "police"
        cfg["shirt"] = Color(0.14, 0.19, 0.36)
        cfg["pants"] = Color(0.10, 0.14, 0.28)
        cfg["shoes"] = Color(0.07, 0.07, 0.08)
        cfg["cap"] = Color(0.11, 0.15, 0.30)
        cfg["glasses"] = false
        cfg["scale"] = rng.randf_range(1.0, 1.08)
    elif kind == 1:
        cfg["id"] = "elderly"
        cfg["shirt"] = [Color(0.78, 0.74, 0.62), Color(0.55, 0.58, 0.46)][rng.randi() % 2]
        cfg["pants"] = Color(0.42, 0.40, 0.37)
        cfg["hair"] = Color(0.82, 0.82, 0.80)
        cfg["beard"] = true
        cfg["stoop"] = -0.26
        cfg["scale"] = rng.randf_range(0.88, 0.96)
        cfg["speed"] = rng.randf_range(0.72, 0.9)
    elif kind == 2:
        cfg["id"] = "teacher"
        cfg["shirt"] = [Color(0.85, 0.86, 0.88), Color(0.36, 0.52, 0.62)][rng.randi() % 2]
        cfg["pants"] = Color(0.22, 0.24, 0.30)
        cfg["glasses"] = true
    else:
        cfg["id"] = "vendor"
        cfg["shirt"] = [Color(0.83, 0.45, 0.20), Color(0.32, 0.46, 0.34), Color(0.72, 0.24, 0.26)][rng.randi() % 3]
        cfg["pants"] = Color(0.33, 0.30, 0.26)
        cfg["cap"] = Color(0.86, 0.82, 0.72) if rng.randf() < 0.5 else Color(0, 0, 0, 0)
    return cfg

static func _mat(c: Color) -> StandardMaterial3D:
    var m := StandardMaterial3D.new()
    m.albedo_color = c
    m.roughness = 1.0
    m.metallic = 0.0
    return m

static func _mesh(parent: Node3D, nm: String, m: Mesh, pos: Vector3, mat: Material) -> MeshInstance3D:
    var mi := MeshInstance3D.new()
    mi.name = nm
    mi.mesh = m
    mi.position = pos
    mi.material_override = mat
    parent.add_child(mi)
    return mi

static func _box(s: Vector3) -> BoxMesh:
    var b := BoxMesh.new()
    b.size = s
    return b

static func _joint(parent: Node3D, nm: String, pos: Vector3) -> Node3D:
    var j := Node3D.new()
    j.name = nm
    j.position = pos
    parent.add_child(j)
    return j

static func _build(cfg: Dictionary) -> MeshInstance3D:
    var w: float = cfg["width"]
    var skin: Material = _mat(cfg["skin"])
    var shirt: Material = _mat(cfg["shirt"])
    var pants: Material = _mat(cfg["pants"])
    var shoes: Material = _mat(cfg["shoes"])
    var hair: Material = _mat(cfg["hair"])
    var carry: bool = cfg["carry"]

    var root := MeshInstance3D.new()
    root.name = "Character"
    root.scale = Vector3.ONE * float(cfg["scale"])

    var rig := _joint(root, "Rig", Vector3.ZERO)
    var hips := _joint(rig, "Hips", Vector3(0, HIP_Y, 0))

    var torso := _joint(hips, "Torso", Vector3.ZERO)
    torso.rotation.x = TORSO_LEAN + float(cfg["stoop"])
    _mesh(torso, "Pelvis", _box(Vector3(0.30 * w, 0.17, 0.20 * w)), Vector3(0, 0.085, 0), pants)
    _mesh(torso, "Chest", _box(Vector3(0.36 * w, 0.35, 0.22 * w)), Vector3(0, 0.325, 0), shirt)

    var neck := _joint(torso, "Neck", Vector3(0, TORSO_LEN, 0))
    _mesh(neck, "NeckMesh", _box(Vector3(0.10, 0.09, 0.10)), Vector3(0, 0.045, 0), skin)
    var head := _joint(neck, "Head", Vector3(0, 0.115, 0))
    var skull := SphereMesh.new()
    skull.radius = 0.125
    skull.height = 0.26
    skull.radial_segments = 10
    skull.rings = 5
    _mesh(head, "Skull", skull, Vector3(0, 0.12, 0), skin)
    _mesh(head, "Hair", _box(Vector3(0.235, 0.085, 0.235)), Vector3(0, 0.205, 0.005), hair)
    if cfg["beard"]:
        _mesh(head, "Beard", _box(Vector3(0.17, 0.10, 0.13)), Vector3(0, 0.055, -0.075), hair)
    if cfg["glasses"]:
        var lens := TorusMesh.new()
        lens.inner_radius = 0.030
        lens.outer_radius = 0.048
        lens.rings = 10
        lens.ring_segments = 5
        var frame := _mat(Color(0.10, 0.09, 0.08))
        for side in [-1.0, 1.0]:
            var g := _mesh(head, "Lens%s" % ("L" if side < 0 else "R"), lens, Vector3(side * 0.055, 0.135, -0.115), frame)
            g.rotation.x = PI * 0.5
    var cap_col: Color = cfg["cap"]
    if cap_col.a > 0.0:
        var crown := CylinderMesh.new()
        crown.top_radius = 0.125
        crown.bottom_radius = 0.132
        crown.height = 0.085
        crown.radial_segments = 10
        crown.rings = 1
        _mesh(head, "Cap", crown, Vector3(0, 0.245, 0.005), _mat(cap_col))
        _mesh(head, "CapBrim", _box(Vector3(0.24, 0.025, 0.14)), Vector3(0, 0.205, -0.13), _mat(cap_col.darkened(0.45)))

    var band_col: Color = cfg["band"]
    for side in [-1.0, 1.0]:
        var tag := "L" if side < 0 else "R"
        var sh := _joint(torso, "Shoulder%s" % tag, Vector3(side * SHOULDER_SPREAD * w, 0.47, 0))
        sh.rotation = Vector3(CARRY_ARM_PITCH, side * CARRY_ARM_YAW, 0) if carry else Vector3(0.1, 0, side * 0.09)
        _mesh(sh, "UpperArm%s" % tag, _box(Vector3(0.105, UPPER_ARM, 0.105)), Vector3(0, -UPPER_ARM * 0.5, 0), shirt)
        var el := _joint(sh, "Elbow%s" % tag, Vector3(0, -UPPER_ARM, 0))
        el.rotation.x = CARRY_ELBOW if carry else SWING_ELBOW
        _mesh(el, "Forearm%s" % tag, _box(Vector3(0.095, FOREARM, 0.095)), Vector3(0, -FOREARM * 0.5, 0), skin)
        _mesh(el, "Hand%s" % tag, _box(Vector3(0.095, 0.10, 0.09)), Vector3(0, -FOREARM - 0.05, 0), skin)
        if band_col.a > 0.0:
            _mesh(el, "Band%s" % tag, _box(Vector3(0.105, 0.045, 0.105)), Vector3(0, -FOREARM + 0.02, 0), _mat(band_col))

    for side in [-1.0, 1.0]:
        var tag := "L" if side < 0 else "R"
        var leg := _joint(hips, "Leg%s" % tag, Vector3(side * HIP_SPREAD * w, 0, 0))
        _mesh(leg, "Thigh%s" % tag, _box(Vector3(0.155, THIGH_LEN, 0.165)), Vector3(0, -THIGH_LEN * 0.5, 0), pants)
        var knee := _joint(leg, "Knee%s" % tag, Vector3(0, -THIGH_LEN, 0))
        _mesh(knee, "Shin%s" % tag, _box(Vector3(0.13, SHIN_LEN, 0.14)), Vector3(0, -SHIN_LEN * 0.5, 0), pants)
        var ankle := _joint(knee, "Ankle%s" % tag, Vector3(0, -SHIN_LEN, 0))
        _mesh(ankle, "Foot%s" % tag, _box(Vector3(0.125, 0.09, 0.27)), Vector3(0, -0.045, -0.055), shoes)

    var cargo := _joint(torso, "Carry", Vector3(0, 0.315, -0.27))
    if carry:
        _build_sixpack(cargo)

    var ap := AnimationPlayer.new()
    ap.name = "Anim"
    var lib := AnimationLibrary.new()
    lib.add_animation("run", _anim_run(carry, float(cfg["stoop"])))
    lib.add_animation("jump", _anim_jump(carry, float(cfg["stoop"])))
    lib.add_animation("stumble", _anim_stumble(carry, float(cfg["stoop"])))
    lib.add_animation("idle", _anim_idle(carry, float(cfg["stoop"])))
    ap.add_animation_library("", lib)
    root.add_child(ap)
    return root

static func _build_sixpack(parent: Node3D) -> void:
    var pack := _joint(parent, "SixPack", Vector3.ZERO)
    pack.rotation.x = -0.18
    var water := _mat(Color(0.58, 0.85, 0.95))
    var cap := _mat(Color(0.20, 0.42, 0.72))
    var wrap := _mat(Color(0.94, 0.94, 0.92))
    _mesh(pack, "Carton", _box(Vector3(0.64, 0.11, 0.19)), Vector3(0, -0.082, 0), wrap)
    var body := CylinderMesh.new()
    body.top_radius = 0.064
    body.bottom_radius = 0.070
    body.height = 0.23
    body.radial_segments = 8
    body.rings = 1
    var neck := CylinderMesh.new()
    neck.top_radius = 0.034
    neck.bottom_radius = 0.039
    neck.height = 0.062
    neck.radial_segments = 6
    neck.rings = 1
    for i in 6:
        var slot: Vector2 = PACK_SLOTS[i]
        var b := _joint(pack, "Bottle%d" % i, Vector3(slot.x * 0.222, 0.060, (slot.y - 0.5) * 0.165))
        _mesh(b, "Body", body, Vector3.ZERO, water)
        _mesh(b, "Cap", neck, Vector3(0, 0.142, 0), cap)

static func _track(a: Animation, path: String, keys: Array) -> void:
    var t := a.add_track(Animation.TYPE_VALUE)
    a.track_set_path(t, NodePath(path))
    a.track_set_interpolation_type(t, Animation.INTERPOLATION_LINEAR)
    a.value_track_set_update_mode(t, Animation.UPDATE_CONTINUOUS)
    for k in keys:
        a.track_insert_key(t, k[0], k[1])

static func _rot(x: float) -> Vector3:
    return Vector3(x, 0, 0)

static func _arm_keys(carry: bool, yaw: float, phase: float) -> Array:
    if carry:
        var p := CARRY_ARM_PITCH
        return [
            [0.0, Vector3(p + 0.03 * phase, yaw, 0)],
            [HALF * 0.5, Vector3(p - 0.04 * phase, yaw, 0)],
            [HALF, Vector3(p + 0.03 * phase, yaw, 0)],
            [HALF * 1.5, Vector3(p - 0.04 * phase, yaw, 0)],
            [RUN_LEN, Vector3(p + 0.03 * phase, yaw, 0)],
        ]
    return [
        [0.0, Vector3(-0.62 * phase, 0, 0)],
        [HALF * 0.5, Vector3(0.0, 0, 0)],
        [HALF, Vector3(0.62 * phase, 0, 0)],
        [HALF * 1.5, Vector3(0.0, 0, 0)],
        [RUN_LEN, Vector3(-0.62 * phase, 0, 0)],
    ]

static func _anim_run(carry: bool, stoop: float) -> Animation:
    var a := Animation.new()
    a.length = RUN_LEN
    a.loop_mode = Animation.LOOP_LINEAR
    var lean := TORSO_LEAN + stoop

    _track(a, "Rig:position", [
        [0.0, Vector3(0, 0.0, 0)], [0.14, Vector3(0, 0.055, 0)],
        [HALF, Vector3(0, 0.0, 0)], [0.42, Vector3(0, 0.055, 0)],
        [RUN_LEN, Vector3(0, 0.0, 0)],
    ])
    _track(a, "Rig/Hips:rotation", [
        [0.0, Vector3(0, 0.10, 0.045)], [HALF, Vector3(0, -0.10, -0.045)],
        [RUN_LEN, Vector3(0, 0.10, 0.045)],
    ])
    _track(a, "Rig/Hips/Torso:rotation", [
        [0.0, Vector3(lean, -0.07, 0)], [0.14, Vector3(lean - 0.04, 0, 0)],
        [HALF, Vector3(lean, 0.07, 0)], [0.42, Vector3(lean - 0.04, 0, 0)],
        [RUN_LEN, Vector3(lean, -0.07, 0)],
    ])
    _track(a, "Rig/Hips/Torso/Neck:rotation", [
        [0.0, _rot(-lean * 0.55)], [0.14, _rot(-lean * 0.55 + 0.05)],
        [HALF, _rot(-lean * 0.55)], [0.42, _rot(-lean * 0.55 + 0.05)],
        [RUN_LEN, _rot(-lean * 0.55)],
    ])

    var thigh := [0.55, 0.00, -0.50, 0.25, 0.55]
    var knee := [-0.15, -0.45, -0.25, -1.30, -0.15]
    var ankle := [0.10, -0.10, 0.35, -0.20, 0.10]
    var times := [0.0, 0.14, HALF, 0.42, RUN_LEN]
    for side in [-1.0, 1.0]:
        var tag := "L" if side < 0 else "R"
        var off := 0 if side < 0 else 2
        var kt := []
        var kk := []
        var ka := []
        for i in 5:
            var j := (i + off) % 4
            kt.append([times[i], _rot(thigh[j])])
            kk.append([times[i], _rot(knee[j])])
            ka.append([times[i], _rot(ankle[j])])
        _track(a, "Rig/Hips/Leg%s:rotation" % tag, kt)
        _track(a, "Rig/Hips/Leg%s/Knee%s:rotation" % [tag, tag], kk)
        _track(a, "Rig/Hips/Leg%s/Knee%s/Ankle%s:rotation" % [tag, tag, tag], ka)
        _track(a, "Rig/Hips/Torso/Shoulder%s:rotation" % tag, _arm_keys(carry, side * CARRY_ARM_YAW, side))
    return a

static func _anim_jump(carry: bool, stoop: float) -> Animation:
    var a := Animation.new()
    a.length = 0.62
    a.loop_mode = Animation.LOOP_NONE
    var lean := TORSO_LEAN + stoop
    _track(a, "Rig:position", [[0.0, Vector3(0, 0.03, 0)], [0.18, Vector3(0, 0.075, 0)], [0.62, Vector3(0, 0.05, 0)]])
    _track(a, "Rig/Hips:rotation", [[0.0, Vector3.ZERO], [0.62, Vector3.ZERO]])
    _track(a, "Rig/Hips/Torso:rotation", [[0.0, _rot(lean * 0.5)], [0.2, _rot(lean * 0.35)], [0.62, _rot(lean * 0.8)]])
    _track(a, "Rig/Hips/LegL:rotation", [[0.0, _rot(0.35)], [0.2, _rot(1.05)], [0.62, _rot(0.55)]])
    _track(a, "Rig/Hips/LegL/KneeL:rotation", [[0.0, _rot(-0.5)], [0.2, _rot(-1.45)], [0.62, _rot(-0.8)]])
    _track(a, "Rig/Hips/LegR:rotation", [[0.0, _rot(-0.15)], [0.2, _rot(-0.55)], [0.62, _rot(-0.25)]])
    _track(a, "Rig/Hips/LegR/KneeR:rotation", [[0.0, _rot(-0.35)], [0.2, _rot(-0.7)], [0.62, _rot(-0.45)]])
    for side in [-1.0, 1.0]:
        var tag := "L" if side < 0 else "R"
        var base: float = CARRY_ARM_PITCH if carry else 0.0
        var yaw: float = float(side) * CARRY_ARM_YAW if carry else 0.0
        _track(a, "Rig/Hips/Torso/Shoulder%s:rotation" % tag, [
            [0.0, Vector3(base, yaw, 0)],
            [0.2, Vector3(base + (0.12 if carry else -0.75), yaw, 0)],
            [0.62, Vector3(base + (0.04 if carry else -0.3), yaw, 0)],
        ])
    return a

static func _anim_stumble(carry: bool, stoop: float) -> Animation:
    var a := Animation.new()
    a.length = 0.75
    a.loop_mode = Animation.LOOP_NONE
    var lean := TORSO_LEAN + stoop
    _track(a, "Rig:position", [[0.0, Vector3.ZERO], [0.18, Vector3(0, -0.13, 0)], [0.45, Vector3(0, -0.04, 0)], [0.75, Vector3.ZERO]])
    _track(a, "Rig/Hips:rotation", [[0.0, Vector3.ZERO], [0.2, Vector3(0, 0.30, 0.22)], [0.5, Vector3(0, -0.16, -0.12)], [0.75, Vector3.ZERO]])
    _track(a, "Rig/Hips/Torso:rotation", [[0.0, _rot(lean)], [0.16, _rot(lean - 0.48)], [0.42, _rot(lean + 0.16)], [0.75, _rot(lean)]])
    _track(a, "Rig/Hips/Torso/Neck:rotation", [[0.0, _rot(-lean * 0.55)], [0.16, _rot(-lean * 0.55 + 0.4)], [0.75, _rot(-lean * 0.55)]])
    _track(a, "Rig/Hips/LegL:rotation", [[0.0, _rot(0.55)], [0.18, _rot(-0.35)], [0.45, _rot(0.85)], [0.75, _rot(0.55)]])
    _track(a, "Rig/Hips/LegL/KneeL:rotation", [[0.0, _rot(-0.15)], [0.18, _rot(-0.95)], [0.45, _rot(-0.35)], [0.75, _rot(-0.15)]])
    _track(a, "Rig/Hips/LegR:rotation", [[0.0, _rot(-0.5)], [0.18, _rot(0.75)], [0.45, _rot(-0.25)], [0.75, _rot(-0.5)]])
    _track(a, "Rig/Hips/LegR/KneeR:rotation", [[0.0, _rot(-0.25)], [0.18, _rot(-1.2)], [0.45, _rot(-0.55)], [0.75, _rot(-0.25)]])
    for side in [-1.0, 1.0]:
        var tag := "L" if side < 0 else "R"
        var base: float = CARRY_ARM_PITCH if carry else 0.0
        var yaw: float = float(side) * CARRY_ARM_YAW if carry else 0.0
        _track(a, "Rig/Hips/Torso/Shoulder%s:rotation" % tag, [
            [0.0, Vector3(base, yaw, 0)],
            [0.16, Vector3(base + (0.35 if carry else 1.15), yaw, side * 0.45)],
            [0.45, Vector3(base - 0.1, yaw, side * 0.12)],
            [0.75, Vector3(base, yaw, 0)],
        ])
    return a

static func _anim_idle(carry: bool, stoop: float) -> Animation:
    var a := Animation.new()
    a.length = 1.8
    a.loop_mode = Animation.LOOP_LINEAR
    var lean := (TORSO_LEAN + stoop) * 0.35
    _track(a, "Rig:position", [[0.0, Vector3.ZERO], [0.9, Vector3(0, 0.018, 0)], [1.8, Vector3.ZERO]])
    _track(a, "Rig/Hips:rotation", [[0.0, Vector3(0, 0.04, 0)], [0.9, Vector3(0, -0.04, 0)], [1.8, Vector3(0, 0.04, 0)]])
    _track(a, "Rig/Hips/Torso:rotation", [[0.0, _rot(lean)], [0.9, _rot(lean - 0.035)], [1.8, _rot(lean)]])
